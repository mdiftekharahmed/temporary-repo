SIGNAL PROCESSING LAB — STANDALONE COPY-PASTE CODES

HOW TO RUN

PYTHON: 1. Install Python 3. 2. Install the two required packages once:
python -m pip install numpy matplotlib (On some Linux systems use:
python3 -m pip install –user numpy matplotlib) 3. Copy ONE complete
Python program into a file, for example exp1.py. 4. Run it with: python
exp1.py

Each Python program below includes its own imports. No external data
files are needed.

MATLAB: 1. Open MATLAB. 2. Copy ONE complete MATLAB program into a new
script. 3. Save it, for example exp1.m, and press Run. All MATLAB
programs use standard MATLAB functions and need no extra toolbox.

IMPORTANT: - Run one experiment at a time. Do not paste multiple
programs into the same script. - Python uses zero-based indexing; MATLAB
uses one-based indexing. - Continuous-time signals are represented
numerically using closely spaced samples. - An ideal impulse cannot be
drawn exactly by a computer; the programs use a unit-area approximation.

==================================================

    import numpy as np
    import matplotlib.pyplot as plt

    # Signal 1: y(t) = cos(t)
    t1 = np.linspace(0, 10, 1000)
    y1 = np.cos(t1)

    plt.ylabel("Amplitude")

================================================== EXPERIMENT 2 —
CONTINUOUS-TIME BASIC SIGNALS
==================================================

PYTHON

    import numpy as np
    import matplotlib.pyplot as plt

    t = np.linspace(-5, 5, 1001)
    dt = t[1] - t[0]

    # Unit step
    unit_step = np.where(t >= 0, 1.0, 0.0)

    # Unit impulse approximation with area approximately equal to 1
    unit_impulse = np.zeros_like(t)
    zero_index = np.argmin(np.abs(t))
    unit_impulse[zero_index] = 1.0 / dt

    # Ramp
    ramp = np.maximum(t, 0)

    # Periodic sinusoidal signal, frequency = 1 Hz
    sinusoid = np.sin(2 * np.pi * t)

    plt.figure(figsize=(10, 8))

    plt.subplot(4, 1, 1)
    plt.plot(t, unit_step)
    plt.title("Continuous-Time Unit Step")
    plt.grid(True)

    plt.subplot(4, 1, 2)
    plt.plot(t, unit_impulse)
    plt.title("Unit Impulse Approximation")
    plt.grid(True)

    plt.subplot(4, 1, 3)
    plt.plot(t, ramp)
    plt.title("Continuous-Time Ramp")
    plt.grid(True)

    plt.subplot(4, 1, 4)
    plt.plot(t, sinusoid)
    plt.title("Continuous-Time Sinusoid")
    plt.grid(True)

    plt.tight_layout()
    plt.show()

================================================== EXPERIMENT 5 —
VERIFICATION OF SYSTEM PROPERTIES
==================================================

PYTHON

    import numpy as np

    n = np.arange(-5, 6)
    x = np.sin(n)

    # Example causal system: y[n] = x[n] + x[n-1]
    def causal_system(signal):
        return signal + np.concatenate(([0], signal[:-1]))

    # Example non-causal system: y[n] = x[n+1]
    def noncausal_system(signal):
        return np.concatenate((signal[1:], [0]))

    # Memoryless system: y[n] = 2x[n]
    def memoryless_system(signal):
        return 2 * signal

    # Dynamic system: y[n] = x[n] + x[n-1]
    def dynamic_system(signal):
        return signal + np.concatenate(([0], signal[:-1]))

    # Linear system: y[n] = 2x[n]
    def linear_system(signal):
        return 2 * signal

    # Nonlinear system: y[n] = x[n]^2
    def nonlinear_system(signal):
        return signal**2

    # Time-invariant system: y[n] = 2x[n]
    def time_invariant_system(signal):
        return 2 * signal

    # Time-variant system: y[n] = n*x[n]
    def time_variant_system(signal):
        return n * signal

    print("Causal system output:")
    print(causal_system(x))

    print("Non-causal system output:")
    print(noncausal_system(x))

    print("Memoryless system output:")
    print(memoryless_system(x))

    print("Dynamic system output:")
    print(dynamic_system(x))

    print("Linear system output:")
    print(linear_system(x))

    print("Nonlinear system output:")
    print(nonlinear_system(x))

    print("Time-invariant system output:")
    print(time_invariant_system(x))

    print("Time-variant system output:")
    print(time_variant_system(x))

    # Test linearity for T(x) = 2x
    x1 = np.array([1.0, 2.0, 3.0])
    x2 = np.array([2.0, 1.0, 4.0])
    a = 2.0
    b = 3.0

    left_side = linear_system(a*x1 + b*x2)
    right_side = a*linear_system(x1) + b*linear_system(x2)

    print("Linearity test passed:", np.allclose(left_side, right_side))

================================================== EXPERIMENT 6 —
CONTINUOUS-TIME CONVOLUTION
==================================================

PYTHON

    import numpy as np
    import matplotlib.pyplot as plt

    dt = 0.01
    t = np.arange(0, 10, dt)

    # Input signals, both zero for negative time
    x1 = np.exp(-t)
    x2 = np.exp(-2*t)

    # Numerical approximation to continuous-time convolution
    y = np.convolve(x1, x2) * dt

    # Time axis for convolution output
    ty = np.arange(len(y)) * dt

    # Analytical result is valid for 0 <= t <= 10
    analytical = np.exp(-ty) - np.exp(-2*ty)
    valid = ty <= 10

    plt.figure(figsize=(10, 8))

    plt.subplot(3, 1, 1)
    plt.plot(t, x1)
    plt.title("First Input: x1(t) = exp(-t)")
    plt.grid(True)

    plt.subplot(3, 1, 2)
    plt.plot(t, x2)
    plt.title("Second Input: x2(t) = exp(-2t)")
    plt.grid(True)

    plt.subplot(3, 1, 3)
    plt.plot(ty, y, label="Numerical convolution")
    plt.plot(ty[valid], analytical[valid], "--", label="Analytical result")
    plt.title("Continuous-Time Convolution")
    plt.xlabel("Time (s)")
    plt.legend()
    plt.grid(True)

    plt.tight_layout()
    plt.show()

MATLAB

    clc;
    clear;
    close all;

    dt = 0.01;
    t = 0:dt:10-dt;

    % Input signals
    x1 = exp(-t);
    x2 = exp(-2*t);

    % Numerical approximation to continuous-time convolution
    y = conv(x1,x2)*dt;

    % Time axis for convolution output
    ty = (0:length(y)-1)*dt;

    % Analytical result is valid for 0 <= t <= 10
    analytical = exp(-ty)-exp(-2*ty);
    valid = ty <= 10;

    figure;

    subplot(3,1,1);
    plot(t,x1);
    title('First Input: x1(t) = exp(-t)');
    grid on;

    subplot(3,1,2);
    plot(t,x2);
    title('Second Input: x2(t) = exp(-2t)');
    grid on;

    subplot(3,1,3);
    plot(ty,y);
    hold on;
    plot(ty(valid),analytical(valid),'--');
    title('Continuous-Time Convolution');
    xlabel('Time (s)');
    legend('Numerical convolution','Analytical result');
    grid on;

================================================== EXPERIMENT 8 —
TIME-DOMAIN AND FREQUENCY-DOMAIN
==================================================

PYTHON

    import numpy as np
    import matplotlib.pyplot as plt

    Fs = 100
    t = np.arange(0, 1, 1/Fs)

    # Signal contains 5 Hz and 10 Hz components
    x = np.sin(2*np.pi*5*t) + 0.5*np.sin(2*np.pi*10*t)

    # Calculate FFT
    N = len(x)
    X = np.fft.fft(x)

    # Positive frequency axis and spectrum
    frequency = np.fft.rfftfreq(N, d=1/Fs)
    spectrum = np.abs(np.fft.rfft(x))/N

    # Convert to one-sided amplitude spectrum
    if N % 2 == 0:
        spectrum[1:-1] *= 2
    else:
        spectrum[1:] *= 2

    plt.figure(figsize=(10, 7))

    plt.subplot(2, 1, 1)
    plt.plot(t, x)
    plt.title("Time-Domain Signal")
    plt.xlabel("Time (seconds)")
    plt.ylabel("Amplitude")
    plt.grid(True)

    plt.subplot(2, 1, 2)
    plt.stem(frequency, spectrum)
    plt.title("One-Sided Amplitude Spectrum")
    plt.xlabel("Frequency (Hz)")
    plt.ylabel("Magnitude")
    plt.xlim(0, Fs/2)
    plt.grid(True)

    plt.tight_layout()
    plt.show()

    import numpy as np
    import matplotlib.pyplot as plt

    A = 1
    T = 2

    # Rectangular pulse in time domain
    t = np.linspace(-3, 3, 1001)

    frequency = np.linspace(-5, 5, 2001)

    plt.plot(frequency, phase)

================================================== EXPERIMENT 10 —
SAMPLING AND ALIASING ==================================================

PYTHON

    import numpy as np
    import matplotlib.pyplot as plt

    signal_frequency = 50

    # High-resolution original signal
    t = np.linspace(0, 0.1, 2000)
    original = np.sin(2*np.pi*signal_frequency*t)

    # Sampling above Nyquist rate
    Fs_good = 200
    t_good = np.arange(0, 0.1, 1/Fs_good)
    x_good = np.sin(2*np.pi*signal_frequency*t_good)

    # Sampling below Nyquist rate
    Fs_low = 80
    t_low = np.arange(0, 0.1, 1/Fs_low)
    x_low = np.sin(2*np.pi*signal_frequency*t_low)

    # Aliased frequency: |50 - 80| = 30 Hz
    # The sampled points correspond to a negative 30 Hz sinusoid
    alias = -np.sin(2*np.pi*30*t)

    plt.figure(figsize=(10, 8))

    plt.subplot(3, 1, 1)
    plt.plot(t, original)

    import numpy as np
    import matplotlib.pyplot as plt

    Fs = 1000

    # Include both t = 0 and t = 10
    t = np.arange(0, 10 + 1/Fs, 1/Fs)

    # Given signal
    x = np.cos(2*np.pi*47*t) + np.cos(2*np.pi*219*t)

    # DFT
    N = len(x)
    X = np.fft.fft(x)

    # One-sided frequency axis
    frequency = np.fft.rfftfreq(N, d=1/Fs)

    # One-sided power spectral density (periodogram)
    P = np.abs(np.fft.rfft(x))**2 / (Fs*N)

    # Double positive-frequency bins, excluding DC.
    # N is odd here, so there is no Nyquist bin.
    P[1:] = 2*P[1:]

    plt.figure(figsize=(10, 7))

    plt.subplot(2, 1, 1)
    plt.plot(t, x)
    plt.title("Sampled Signal")
    plt.xlabel("Time (seconds)")
    plt.ylabel("Amplitude")
    plt.grid(True)

    plt.subplot(2, 1, 2)
    plt.plot(frequency, P)
    plt.title("One-Sided Power Spectral Density")
    plt.xlabel("Frequency (Hz)")
    plt.ylabel("Power/Hz")
    plt.xlim(0, 300)
    plt.grid(True)

    plt.tight_layout()
    plt.show()

    print("Sampling frequency:", Fs, "Hz")
    print("Number of samples:", N)
    print("Frequency resolution:", Fs/N, "Hz")
    print("Expected frequency components: 47 Hz and 219 Hz")
